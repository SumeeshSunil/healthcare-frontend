src/redux/slices/appointmentSlice.js
Clinical Appointments Lifecycle
Manages the appointment queue and status transitions across patient, admin, and doctor workflows.
Exports `addAppointment` (pending), `approveAppointment` (confirmed), `rejectAppointment` (cancelled), `cancelAppointment`, and `completeAppointment` (completed).
import { createSlice } from "@reduxjs/toolkit";
import initialAppointments from "../../data/dummyAppointments.json";

const appointmentSlice = createSlice({
    name: "appointment",
    initialState: { appointments: initialAppointments },
    reducers: {
        addAppointment: (state, action) => { state.appointments.unshift(action.payload); },
        approveAppointment: (state, action) => {
            const app = state.appointments.find(a => a.id === action.payload);
            if (app) app.status = "confirmed";
        },
        completeAppointment: (state, action) => {
            const app = state.appointments.find(a => a.id === action.payload);
            if (app) app.status = "completed";
        }
    }
});
src/redux/slices/billingSlice.js
Financial Billing & Invoices State
Manages clinical invoices, amounts due, payment status, and payment settlement records.
Exports `addBill` and `payBill` (with normalized payload handling supporting both object `{ billId, paymentMethod }` and primitive ID numbers).
import { createSlice } from "@reduxjs/toolkit";
import dummyBills from "../../data/dummyBills.json";

const billingSlice = createSlice({
    name: "billing",
    initialState: { bills: dummyBills },
    reducers: {
        addBill: (state, action) => { state.bills.unshift(action.payload); },
        payBill: (state, action) => {
            const payload = action.payload;
            const billId = typeof payload === "object" ? payload.billId : payload;
            const paymentMethod = typeof payload === "object" ? payload.paymentMethod : "Online UPI";
            const bill = state.bills.find((b) => b.id === Number(billId));
            if (bill) {
                bill.status = "paid";
                bill.paidOn = new Date().toISOString().split("T")[0];
                bill.paymentMethod = paymentMethod;
            }
        }
    }
});

3 Shared UI Layout & Context Components

src/components/Toast.jsx
Context API Toast Provider
Provides lightweight, auto-dismissing toast notifications across the app.
Uses React Context API (`createContext`, `useContext`, `useCallback`). Exports `useToast()` with `toast.success()`, `toast.error()`, `toast.warning()`, `toast.info()`.
import { createContext, useContext, useState, useCallback } from "react";

const ToastContext = createContext();

export function ToastProvider({ children }) {
    const [toasts, setToasts] = useState([]);

    const removeToast = useCallback((id) => {
        setToasts((prev) => prev.filter((t) => t.id !== id));
    }, []);

    const addToast = useCallback((message, type = "info", title = "") => {
        const id = Date.now();
        setToasts((prev) => [...prev, { id, message, type, title }]);
        setTimeout(() => removeToast(id), 4000);
    }, [removeToast]);

    return (
        <ToastContext.Provider value={{ addToast }}>
            {children}
        </ToastContext.Provider>
    );
}

4 Security & Routing System

src/routes/ProtectedRoute.jsx
Role-Based Route Guard
Prevents unauthorized access to protected pages.
Checks `isAuthenticated` (redirects to `/login` if false) and checks `allowedRoles` (redirects to `/unauthorized` if role is forbidden).
import { useSelector } from "react-redux";
import { Navigate, Outlet } from "react-router-dom";

function ProtectedRoute({ allowedRoles }) {
    const { isAuthenticated, user } = useSelector((state) => state.auth);

    if (!isAuthenticated) {
        return <Navigate to="/login" replace />;
    }

    if (allowedRoles && !allowedRoles.includes(user?.role)) {
        return <Navigate to="/unauthorized" replace />;
    }

    return <Outlet />;
}

export default ProtectedRoute;
